server {
    # 1. Listen on Port 80 (HTTP) and redirect to HTTPS
    listen 80;
    listen [::]:80;
    server_name inceasar.com www.inceasar.com;

    return 301 https://$host$request_uri;
}

server {
    # 2. Listen on Port 443 (HTTPS) with HTTP/2 enabled
    listen 443 ssl http2;
    listen [::]:443 ssl http2;
    server_name inceasar.com www.inceasar.com;

    # 3. SSL Configuration (Managed by Certbot)
    ssl_certificate /etc/letsencrypt/live/inceasar.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/inceasar.com/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;

    # 4. Upload Size & Client Timeouts
    # (Settings you requested)
    client_max_body_size 200M;
    client_body_timeout 300s;

    # Enable Gzip compression for speed
    gzip on;
    gzip_types text/plain text/css application/json application/javascript text/xml application/xml application/xml+rss text/javascript;

    # 5. Proxy Configuration
    location / {
        proxy_pass http://localhost:3000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;

        # Forward real IP
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # Backend Timeouts (Settings you requested)
        proxy_read_timeout 300s;
        proxy_send_timeout 300s;
    }
}
